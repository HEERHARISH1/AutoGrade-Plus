Programming Assignment #1 - Factorial Function

def factorial(n):
    """
    Calculate factorial of n using recursion.
    
    Args:
        n: Non-negative integer
        
    Returns:
        Factorial of n
    """
    # Handle edge cases
    if n == 0 or n == 1:
        return 1
    
    # Recursive case
    return n * factorial(n - 1)


# Test cases to demonstrate correctness
